Skrip Google Apps Script (`Code.gs`)

Salin skrip di bawah ini ke Editor Google Apps Script spreadsheet Anda untuk mengaktifkan Web App API secara langsung.

/* ====================================================================
   BACKEND GOOGLE APPS SCRIPT - SISTEM INFORMASI MUTASI SISWA
   Atur Spreadsheet: Buat 4 Sheet: "Users", "Data_Utama", "Activity_Logs", "Settings"
   ==================================================================== */

function doGet(e) {
  return handleRequest(e);
}

function doPost(e) {
  return handleRequest(e);
}

function handleRequest(e) {
  var response = { status: "error", message: "Invalid Action" };
  try {
    var params = e.parameter;
    var postData = e.postData ? JSON.parse(e.postData.contents) : {};
    var action = params.action || postData.action;

    var ss = SpreadsheetApp.getActiveSpreadsheet();

    if (action === "setup") {
      setupDatabaseSheets(ss);
      response = { status: "success", message: "Database sheets initialized!" };
    } else if (action === "get_all") {
      response = {
        status: "success",
        users: getSheetData(ss, "Users"),
        data_utama: getSheetData(ss, "Data_Utama"),
        logs: getSheetData(ss, "Activity_Logs"),
        settings: getSheetData(ss, "Settings")
      };
    } else if (action === "add_mutasi") {
      appendSheetRow(ss, "Data_Utama", postData.data);
      addLog(ss, postData.username, "Menambah Data Mutasi: " + postData.data.Nama_Siswa);
      response = { status: "success" };
    } else if (action === "update_mutasi") {
      updateSheetRow(ss, "Data_Utama", "ID_Data", postData.data.ID_Data, postData.data);
      addLog(ss, postData.username, "Mengubah Data Mutasi ID: " + postData.data.ID_Data);
      response = { status: "success" };
    } else if (action === "delete_mutasi") {
      softDeleteSheetRow(ss, "Data_Utama", "ID_Data", postData.id_data);
      addLog(ss, postData.username, "Soft Delete Data Mutasi ID: " + postData.id_data);
      response = { status: "success" };
    } else if (action === "add_user") {
      appendSheetRow(ss, "Users", postData.data);
      addLog(ss, postData.username, "Menambah User Baru: " + postData.data.Username);
      response = { status: "success" };
    } else if (action === "delete_user") {
      deleteSheetRow(ss, "Users", "ID", postData.id_user);
      addLog(ss, postData.username, "Menghapus User ID: " + postData.id_user);
      response = { status: "success" };
    } else if (action === "update_settings") {
      saveSettingsData(ss, postData.settings);
      addLog(ss, postData.username, "Memperbarui Pengaturan Aplikasi");
      response = { status: "success" };
    }
  } catch (err) {
    response = { status: "error", message: err.toString() };
  }

  return ContentService.createTextOutput(JSON.stringify(response))
    .setMimeType(ContentService.MimeType.JSON);
}

function getSheetData(ss, sheetName) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  var headers = data[0];
  var result = [];
  for (var i = 1; i < data.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = data[i][j];
    }
    result.push(obj);
  }
  return result;
}

function appendSheetRow(ss, sheetName, rowObj) {
  var sheet = ss.getSheetByName(sheetName);
  var headers = sheet.getDataRange().getValues()[0];
  var newRow = [];
  for (var i = 0; i < headers.length; i++) {
    newRow.push(rowObj[headers[i]] || "");
  }
  sheet.appendRow(newRow);
}

function updateSheetRow(ss, sheetName, keyCol, keyVal, rowObj) {
  var sheet = ss.getSheetByName(sheetName);
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var keyIdx = headers.indexOf(keyCol);
  for (var i = 1; i < data.length; i++) {
    if (data[i][keyIdx] == keyVal) {
      for (var j = 0; j < headers.length; j++) {
        if (rowObj[headers[j]] !== undefined) {
          sheet.getRange(i + 1, j + 1).setValue(rowObj[headers[j]]);
        }
      }
      break;
    }
  }
}

function softDeleteSheetRow(ss, sheetName, keyCol, keyVal) {
  var sheet = ss.getSheetByName(sheetName);
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var keyIdx = headers.indexOf(keyCol);
  var statusIdx = headers.indexOf("Status");
  for (var i = 1; i < data.length; i++) {
    if (data[i][keyIdx] == keyVal) {
      sheet.getRange(i + 1, statusIdx + 1).setValue("Nonaktif");
      break;
    }
  }
}

function deleteSheetRow(ss, sheetName, keyCol, keyVal) {
  var sheet = ss.getSheetByName(sheetName);
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var keyIdx = headers.indexOf(keyCol);
  for (var i = 1; i < data.length; i++) {
    if (data[i][keyIdx] == keyVal) {
      sheet.deleteRow(i + 1);
      break;
    }
  }
}

function addLog(ss, username, action) {
  var sheet = ss.getSheetByName("Activity_Logs");
  if (!sheet) return;
  sheet.appendRow([
    "LOG-" + new Date().getTime(),
    username || "System",
    action,
    new Date().toLocaleString("id-ID")
  ]);
}
